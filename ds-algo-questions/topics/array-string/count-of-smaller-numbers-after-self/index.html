Count of Smaller Numbers After Self

Problem

Given an integer array nums, return an array counts where counts[i] is the number of elements to the right of nums[i] that are smaller than it.

ExampleInput: nums = [5,2,6,1] Output: [2,1,1,0]

Approach

A modified merge sort counts these values while sorting index positions instead of the values directly. During each merge step, whenever an element from the right half gets placed before elements still remaining in the left half, every one of those pending left-half elements has exactly that many smaller elements to its right -- so their counts get bumped by however many right-half elements were placed first.

Time O(n log n)Space O(n)
Java
public List<Integer> countSmaller(int[] nums) {
    int n = nums.length;
    Integer[] indices = new Integer[n];
    int[] counts = new int[n];
    for (int i = 0; i < n; i++) {
        indices[i] = i;
    }

    mergeSort(nums, indices, counts, 0, n - 1);

    List<Integer> result = new ArrayList<>();
    for (int c : counts) {
        result.add(c);
    }
    return result;
}

private void mergeSort(int[] nums, Integer[] indices, int[] counts, int lo, int hi) {
    if (lo >= hi) {
        return;
    }
    int mid = lo + (hi - lo) / 2;
    mergeSort(nums, indices, counts, lo, mid);
    mergeSort(nums, indices, counts, mid + 1, hi);
    merge(nums, indices, counts, lo, mid, hi);
}

private void merge(int[] nums, Integer[] indices, int[] counts, int lo, int mid, int hi) {
    Integer[] merged = new Integer[hi - lo + 1];
    int i = lo, j = mid + 1, k = 0;
    int rightCount = 0;

    while (i <= mid && j <= hi) {
        if (nums[indices[j]] < nums[indices[i]]) {
            rightCount++;
            merged[k++] = indices[j++];
        } else {
            counts[indices[i]] += rightCount;
            merged[k++] = indices[i++];
        }
    }
    while (i <= mid) {
        counts[indices[i]] += rightCount;
        merged[k++] = indices[i++];
    }
    while (j <= hi) {
        merged[k++] = indices[j++];
    }

    System.arraycopy(merged, 0, indices, lo, merged.length);
}
