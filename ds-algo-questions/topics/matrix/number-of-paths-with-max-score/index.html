Number of Paths with Max Score

Problem

Given a square board of digits (and 'S'/'E' start/end markers, 'X' obstacles), moving from bottom-right to top-left via up/left/up-left, find the maximum sum obtainable and the number of paths achieving that sum, modulo 10^9+7.

ExampleInput: board = ["E23","2X2","12S"] Output: [7,1]

Approach

DP tracking a pair (bestSum, pathCount) per cell. Each cell combines results from its 3 predecessors (down, right, down-right): take the maximum bestSum among reachable predecessors, and sum the pathCounts of every predecessor that ties that maximum -- obstacle cells are simply unreachable.

Time O(n^2)Space O(n^2)
Java
public int[] pathsWithMaxScore(List<String> board) {
    int n = board.size();
    long MOD = 1_000_000_007;
    long[][] sum = new long[n][n];
    long[][] count = new long[n][n];

    char[][] grid = new char[n][n];
    for (int i = 0; i < n; i++) {
        grid[i] = board.get(i).toCharArray();
    }

    count[n - 1][n - 1] = 1;
    grid[0][0] = '0';
    grid[n - 1][n - 1] = '0';

    for (int r = n - 1; r >= 0; r--) {
        for (int c = n - 1; c >= 0; c--) {
            if (grid[r][c] == 'X' || (r == n - 1 && c == n - 1)) {
                continue;
            }

            long bestSum = -1;
            long bestCount = 0;

            int[][] preds = {{r + 1, c}, {r, c + 1}, {r + 1, c + 1}};
            for (int[] p : preds) {
                int pr = p[0], pc = p[1];
                if (pr >= n || pc >= n || (count[pr][pc] == 0 && !(pr == n - 1 && pc == n - 1))) {
                    continue;
                }
                if (sum[pr][pc] > bestSum || (pr == n - 1 && pc == n - 1 && bestSum == -1)) {
                    bestSum = sum[pr][pc];
                    bestCount = count[pr][pc];
                } else if (sum[pr][pc] == bestSum) {
                    bestCount = (bestCount + count[pr][pc]) % MOD;
                }
            }

            if (bestCount > 0) {
                sum[r][c] = bestSum + (grid[r][c] - '0');
                count[r][c] = bestCount;
            }
        }
    }

    if (count[0][0] == 0) {
        return new int[]{0, 0};
    }
    return new int[]{(int) sum[0][0], (int) count[0][0]};
}
