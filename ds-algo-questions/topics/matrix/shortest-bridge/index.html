Shortest Bridge

Problem

Given a grid with exactly two islands, return the minimum number of 0s that must be flipped to 1 to connect them (equivalently, the minimum bridge length between the two islands).

ExampleInput: grid = [[0,1,0],[0,0,0],[0,0,1]] Output: 2

Approach

First, DFS from any land cell to find and mark the entirety of the first island, collecting all of its cells as BFS starting points. Then run a multi-source BFS from those cells outward over water, expanding level by level until a cell belonging to the second island is reached -- the number of BFS levels taken is the bridge length.

Time O(mn)Space O(mn)
Java
public int shortestBridge(int[][] grid) {
    int rows = grid.length, cols = grid[0].length;
    boolean[][] visited = new boolean[rows][cols];
    Queue<int[]> queue = new LinkedList<>();

    outer:
    for (int r = 0; r < rows; r++) {
        for (int c = 0; c < cols; c++) {
            if (grid[r][c] == 1) {
                dfsMark(grid, visited, r, c, queue);
                break outer;
            }
        }
    }

    int[][] dirs = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};
    int steps = 0;

    while (!queue.isEmpty()) {
        int size = queue.size();
        for (int i = 0; i < size; i++) {
            int[] cell = queue.poll();
            for (int[] d : dirs) {
                int nr = cell[0] + d[0], nc = cell[1] + d[1];
                if (nr < 0 || nr >= rows || nc < 0 || nc >= cols || visited[nr][nc]) {
                    continue;
                }
                if (grid[nr][nc] == 1) {
                    return steps;
                }
                visited[nr][nc] = true;
                queue.offer(new int[]{nr, nc});
            }
        }
        steps++;
    }

    return -1;
}

private void dfsMark(int[][] grid, boolean[][] visited, int r, int c, Queue<int[]> queue) {
    if (r < 0 || r >= grid.length || c < 0 || c >= grid[0].length
            || visited[r][c] || grid[r][c] == 0) {
        return;
    }

    visited[r][c] = true;
    queue.offer(new int[]{r, c});

    dfsMark(grid, visited, r + 1, c, queue);
    dfsMark(grid, visited, r - 1, c, queue);
    dfsMark(grid, visited, r, c + 1, queue);
    dfsMark(grid, visited, r, c - 1, queue);
}
