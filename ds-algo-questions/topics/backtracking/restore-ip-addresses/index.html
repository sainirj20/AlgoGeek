Restore IP Addresses

Problem

Given a string containing only digits, return all possible valid IP address combinations that can be formed by inserting dots (each segment 0-255, no leading zeros except a lone "0").

ExampleInput: s = "25525511135" Output: ["255.255.11.135","255.255.111.35"]

Approach

Backtracking over 4 segments: at each step, try taking 1, 2, or 3 characters as the next segment, validating it's a number from 0-255 with no invalid leading zero, and recursing on the rest of the string. Succeed only when exactly 4 segments have been formed and the entire string has been consumed.

Time O(1) (bounded by 4 segments x 3 lengths)Space O(1)
Java
public List<String> restoreIpAddresses(String s) {
    List<String> result = new ArrayList<>();
    backtrack(s, 0, new ArrayList<>(), result);
    return result;
}

private void backtrack(String s, int start, List<String> segments, List<String> result) {
    if (segments.size() == 4) {
        if (start == s.length()) {
            result.add(String.join(".", segments));
        }
        return;
    }

    for (int len = 1; len <= 3 && start + len <= s.length(); len++) {
        String segment = s.substring(start, start + len);
        if (isValidSegment(segment)) {
            segments.add(segment);
            backtrack(s, start + len, segments, result);
            segments.remove(segments.size() - 1);
        }
    }
}

private boolean isValidSegment(String segment) {
    if (segment.length() > 1 && segment.charAt(0) == '0') {
        return false;
    }
    int value = Integer.parseInt(segment);
    return value >= 0 && value <= 255;
}
