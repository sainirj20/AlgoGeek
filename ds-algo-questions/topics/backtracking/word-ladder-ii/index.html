Word Ladder II

Problem

Given a beginWord, an endWord, and a word list, return all shortest transformation sequences from beginWord to endWord.

ExampleInput: beginWord = "hit", endWord = "cog", wordList = ["hot","dot","dog","lot","log","cog"] Output: [["hit","hot","dot","dog","cog"],["hit","hot","lot","log","cog"]]
📷Diagram placeholder — paste the array traversal image here (save it into this problem's folder and set diagram in solutions.py).
LeetCode #126

Approach

Two phases: first, BFS level by level from beginWord, recording every word's parents (the words that lead to it on a shortest path) instead of just marking visited, and stop as soon as endWord's level is found. Second, backtrack from endWord to beginWord using the recorded parent map (DFS), building every shortest path in reverse and reversing each before adding it to the result.

Time O(n * L^2 * 26)Space O(n * L)
Java
public List<List<String>> findLadders(String beginWord, String endWord, List<String> wordList) {
    List<List<String>> result = new ArrayList<>();
    Set<String> words = new HashSet<>(wordList);
    if (!words.contains(endWord)) {
        return result;
    }

    Map<String, List<String>> parents = new HashMap<>();
    Set<String> currentLevel = new HashSet<>();
    currentLevel.add(beginWord);
    words.remove(beginWord);
    boolean found = false;

    while (!currentLevel.isEmpty() && !found) {
        words.removeAll(currentLevel);
        Set<String> nextLevel = new HashSet<>();

        for (String word : currentLevel) {
            char[] chars = word.toCharArray();
            for (int pos = 0; pos < chars.length; pos++) {
                char original = chars[pos];
                for (char c = 'a'; c <= 'z'; c++) {
                    chars[pos] = c;
                    String candidate = new String(chars);
                    if (words.contains(candidate)) {
                        nextLevel.add(candidate);
                        parents.computeIfAbsent(candidate, k -> new ArrayList<>()).add(word);
                        if (candidate.equals(endWord)) {
                            found = true;
                        }
                    }
                }
                chars[pos] = original;
            }
        }

        currentLevel = nextLevel;
    }

    if (found) {
        backtrack(endWord, beginWord, parents, new LinkedList<>(), result);
    }

    return result;
}

private void backtrack(String word, String beginWord, Map<String, List<String>> parents,
                        LinkedList<String> path, List<List<String>> result) {
    path.addFirst(word);
    if (word.equals(beginWord)) {
        result.add(new ArrayList<>(path));
    } else {
        for (String parent : parents.getOrDefault(word, new ArrayList<>())) {
            backtrack(parent, beginWord, parents, path, result);
        }
    }
    path.removeFirst();
}
