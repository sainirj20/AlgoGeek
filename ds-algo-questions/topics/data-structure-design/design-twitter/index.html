Design Twitter

Problem

Design a simplified Twitter supporting posting tweets, following/unfollowing users, and retrieving the 10 most recent tweets in a user's news feed (their own tweets plus those of everyone they follow).

ExamplepostTweet(1,5); getNewsFeed(1) -> [5]; follow(1,2); postTweet(2,6); getNewsFeed(1) -> [6,5]

Approach

Store each user's tweets as a list of (timestamp, tweetId), and each user's followees as a set. For getNewsFeed, use a max-heap seeded with the most recent tweet from the user and every followee, repeatedly popping the most recent tweet and pushing that same user's next-most-recent tweet, stopping once 10 tweets are collected.

Time O(k log f) per feed request (k=10, f=followees)Space O(tweets + follows)
Java
class Twitter {
    private int timestamp = 0;
    private Map<Integer, List<int[]>> tweets = new HashMap<>();
    private Map<Integer, Set<Integer>> follows = new HashMap<>();

    public void postTweet(int userId, int tweetId) {
        tweets.computeIfAbsent(userId, k -> new ArrayList<>()).add(new int[]{timestamp++, tweetId});
    }

    public List<Integer> getNewsFeed(int userId) {
        PriorityQueue<int[]> heap = new PriorityQueue<>((a, b) -> b[0] - a[0]);
        Map<Integer, Integer> pointer = new HashMap<>();

        Set<Integer> people = new HashSet<>(follows.getOrDefault(userId, new HashSet<>()));
        people.add(userId);

        for (int person : people) {
            List<int[]> personTweets = tweets.getOrDefault(person, new ArrayList<>());
            int idx = personTweets.size() - 1;
            if (idx >= 0) {
                heap.offer(new int[]{personTweets.get(idx)[0], personTweets.get(idx)[1], person, idx});
            }
        }

        List<Integer> result = new ArrayList<>();
        while (!heap.isEmpty() && result.size() < 10) {
            int[] top = heap.poll();
            result.add(top[1]);

            int person = top[2], idx = top[3] - 1;
            if (idx >= 0) {
                List<int[]> personTweets = tweets.get(person);
                heap.offer(new int[]{personTweets.get(idx)[0], personTweets.get(idx)[1], person, idx});
            }
        }

        return result;
    }

    public void follow(int followerId, int followeeId) {
        if (followerId != followeeId) {
            follows.computeIfAbsent(followerId, k -> new HashSet<>()).add(followeeId);
        }
    }

    public void unfollow(int followerId, int followeeId) {
        if (follows.containsKey(followerId)) {
            follows.get(followerId).remove(followeeId);
        }
    }
}
